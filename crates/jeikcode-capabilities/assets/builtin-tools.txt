# JeikCode 内置工具清单(built-in tools catalog)
# =====================================================
# 用途: 此文件仅供你(用户)查看 JeikCode 内置了哪些工具,
#       以便把工具名加入 config.toml 的「不折叠白名单」:
#
#   [tools.tool_output]
#   max_bytes = 65536
#   no_fold_tools = ["code_explore"]
#
# 白名单效果: 列出的工具输出永不折叠(哪怕超过 max_bytes),
#   直接完整展示, 不需要 fetch_output 手动取回。
#   等价于工具自带的 never_truncate_result() 契约(code_explore 默认已豁免)。
#
# 提示: 名称大小写敏感; 按需批量添加, 逗号分隔。
# =====================================================

## 一、文件与代码操作
run_command           # 执行终端命令(别名 bash，后台命令返回 PID 与端口)
cd                    # 切换工作目录
read                  # 读取文件与目录树(别名 read_file)
write                 # 写入文件(别名 write_file)
edit                  # 精确文本替换编辑(别名 edit_file)
parallel_edit         # 并行编辑多个文件
glob                  # 按 glob 模式找文件
grep                  # 按正则搜索文件内容
todo_write            # 任务清单(别名 todowrite / todo)
task                  # 派生子代理任务
long_bash_keyword_actions # 维护长时间运行命令的超时白名单

## 二、网络
web_fetch             # 抓取网页内容
web_search            # 网页搜索

## 三、代码智能(codeintel 系列)
code_explore          # 语义检索+调用链+目录全景(默认已豁免折叠)

## 四、工具输出恢复
fetch_output          # 按 artifact_id 分片取回被折叠的输出

## 五、交互 / 其他
request_user_input    # 向用户提问
report_finding        # 报告发现项
jeikcode_config       # 查阅配置指南(action="guide")或热重载运行时(action="reload")

## 常见白名单推荐
# 想彻底告别 fetch_output, 建议至少加入:
#   no_fold_tools = ["code_explore"]
# 注意: 白名单是「工具输出不折叠」; 工具内部自带的智能裁剪
#   (code_explore 的 max_files / span 折叠)不受影响, 仍会控制上下文体积。
